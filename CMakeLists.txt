cmake_minimum_required(VERSION 3.20)
project(VisualStudioLauncher VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(LauncherCore src/Core.cpp)
target_include_directories(LauncherCore PUBLIC src)
if(MSVC)
  set_property(TARGET LauncherCore PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(LauncherCore PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(LauncherCore PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(LauncherCoreTests tests/CoreTests.cpp)
  target_link_libraries(LauncherCoreTests PRIVATE LauncherCore)
  if(MSVC)
    set_property(TARGET LauncherCoreTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(LauncherCoreTests PRIVATE /utf-8 /W4)
  endif()
  add_test(NAME LauncherCoreTests COMMAND LauncherCoreTests)
endif()

if(WIN32)
  enable_language(RC)
  add_executable(VisualStudioLauncher WIN32 src/Main.cpp src/Platform.cpp src/Icons.cpp res/Launcher.rc)
  target_compile_definitions(VisualStudioLauncher PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_include_directories(VisualStudioLauncher PRIVATE res)
  target_link_libraries(VisualStudioLauncher PRIVATE LauncherCore comctl32 comdlg32 shell32 shlwapi ole32 propsys uuid gdi32 user32)
  if(MSVC)
    set_property(TARGET VisualStudioLauncher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(VisualStudioLauncher PRIVATE /W4 /utf-8 /permissive-)
    target_link_options(VisualStudioLauncher PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/res/Launcher.manifest")
  elseif(MINGW)
    target_compile_options(VisualStudioLauncher PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(VisualStudioLauncher PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_sources(VisualStudioLauncher PRIVATE res/Manifest.rc)
  endif()
  install(TARGETS VisualStudioLauncher RUNTIME DESTINATION .)
endif()
